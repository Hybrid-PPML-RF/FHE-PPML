cmake_minimum_required(VERSION 3.10.2)
project(fhe_random_forest)
set(CMAKE_CXX_STANDARD 17)

# Fetch GoogleTest headers — required by fhe_client.h (FRIEND_TEST macro).
# Only the headers are needed; we do not build the test binary.
include(FetchContent)
FetchContent_Declare(googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        release-1.12.1
)
set(BUILD_GMOCK       OFF CACHE BOOL "" FORCE)
set(INSTALL_GTEST     OFF CACHE BOOL "" FORCE)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

find_package(SEAL 4.0.0 REQUIRED)

add_executable(fhe_random_forest
    src/main.cpp
    src/soft_if.cpp
    src/soft_if.h
    src/fhe_client.cpp
    src/fhe_client.h
    src/tree_train_server.cpp
    src/tree_train_server.h
    src/tree_eval_server.cpp
    src/tree_eval_server.h
    src/decision_tree_node.h
    src/enc_tree_node.h
    src/data_types.h
    src/utils.h)

if(TARGET SEAL::seal)
    target_link_libraries(fhe_random_forest PRIVATE SEAL::seal gtest)
elseif(TARGET SEAL::seal_shared)
    target_link_libraries(fhe_random_forest PRIVATE SEAL::seal_shared gtest)
else()
    message(FATAL_ERROR "Cannot find SEAL::seal or SEAL::seal_shared")
endif()
